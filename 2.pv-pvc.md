# 🧩 Problem Statement (PersistentVolumeClaim and Deployment Data Persistence)

<details>
<summary>Click to expand the problem statement</summary>

A MariaDB Deployment running in the `mariadb` namespace was accidentally deleted along with its PersistentVolumeClaim. However the underlying PersistentVolume still exists in the cluster with the data intact.

Your task is to re-establish the Deployment while ensuring the existing data is preserved by reusing the available PersistentVolume.

```
[candidate@base]$ ssh <hostname>
```

### 📋 Configure the following:

1. Create a PersistentVolumeClaim named **`mariadb`** in the **`mariadb`** namespace with the following specifications:
   * **Access mode:** `ReadWriteOnce`
   * **Storage request:** `250Mi`
   * Use the **existing PersistentVolume** available in the cluster

2. Edit the MariaDB Deployment file located at **`~/mariadb-deployment.yaml`** to use the PVC created in Step 1

3. Apply the updated Deployment file to the cluster

4. Ensure the MariaDB Deployment is **running and stable**

</details>

# 🛠 Setup Practice Scenario

### 📥 Script : setup_q02_pvc.sh

```bash
#!/bin/bash
# ============================================================
# CKA Practice | q02: PVC and Deployment Data Persistence
# setup_q02_pvc.sh
# Creates StorageClass, PV, deploys MariaDB, injects sample
# data, then simulates accidental deletion of deployment+PVC
# PV is retained with data intact
# Re-run safe: cleans candidate work and re-injects data
# Run on: controlplane node
# ============================================================

set -euo pipefail

NAMESPACE="mariadb"
PV_NAME="mariadb-pv"
PVC_NAME="mariadb-init"
DEPLOYMENT="mariadb"
SC_NAME="local-path"
HOST_PATH="/mnt/mariadb-data"
DEPLOYMENT_FILE="$HOME/mariadb-deployment.yaml"

echo ""
echo "==========================================="
echo " CKA q02 — Setting up PVC scenario..."
echo "==========================================="
echo ""

# ── Step 1: Reset candidate work ─────────────────────────────
echo "[1/7] Resetting candidate work from previous attempt..."

# Delete candidate-created resources
kubectl delete deployment "$DEPLOYMENT" -n "$NAMESPACE" --ignore-not-found
kubectl delete pvc mariadb -n "$NAMESPACE" --ignore-not-found
kubectl delete pvc "$PVC_NAME" -n "$NAMESPACE" --ignore-not-found

# Remove candidate YAML files
rm -f "$HOME/mariadb-pvc.yaml"
rm -f "$DEPLOYMENT_FILE"

# Patch PV claimRef if stuck in Released state
if kubectl get pv "$PV_NAME" &>/dev/null; then
    PV_STATUS=$(kubectl get pv "$PV_NAME" \
        -o jsonpath='{.status.phase}' 2>/dev/null || true)
    if [ "$PV_STATUS" = "Released" ] || [ "$PV_STATUS" = "Bound" ]; then
        kubectl patch pv "$PV_NAME" \
            --type=json \
            -p='[{"op": "remove", "path": "/spec/claimRef"}]' \
            2>/dev/null || true
    fi
fi

echo "  Candidate work cleared."

# ── Step 2: Create StorageClass if not present ────────────────
echo "[2/7] Checking StorageClass '$SC_NAME'..."
if kubectl get storageclass "$SC_NAME" &>/dev/null; then
    echo "  StorageClass already exists — skipping."
else
    echo "  Creating StorageClass '$SC_NAME'..."
    kubectl apply -f - <<EOF
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: $SC_NAME
provisioner: kubernetes.io/no-provisioner
volumeBindingMode: Immediate
reclaimPolicy: Retain
EOF
    echo "  StorageClass created."
fi

# ── Step 3: Create namespace if not present ───────────────────
echo "[3/7] Checking namespace '$NAMESPACE'..."
if kubectl get namespace "$NAMESPACE" &>/dev/null; then
    echo "  Namespace already exists — skipping."
else
    kubectl create namespace "$NAMESPACE"
    echo "  Namespace created."
fi

# ── Step 4: Recreate PV with fresh hostPath ───────────────────
echo "[4/7] Recreating PersistentVolume '$PV_NAME'..."

# Always delete and recreate PV to ensure fresh data directory
kubectl delete pv "$PV_NAME" --ignore-not-found
kubectl wait --for=delete pv/"$PV_NAME" --timeout=30s 2>/dev/null || true
sudo rm -rf "$HOST_PATH"
sudo mkdir -p "$HOST_PATH"

kubectl apply -f - <<EOF
apiVersion: v1
kind: PersistentVolume
metadata:
  name: $PV_NAME
spec:
  capacity:
    storage: 250Mi
  accessModes:
  - ReadWriteOnce
  persistentVolumeReclaimPolicy: Retain
  storageClassName: $SC_NAME
  hostPath:
    path: $HOST_PATH
EOF
echo "  PersistentVolume created."

# ── Step 5: Create initial PVC and deploy MariaDB ─────────────
echo "[5/7] Creating initial PVC and deploying MariaDB..."

kubectl apply -f - <<EOF
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: $PVC_NAME
  namespace: $NAMESPACE
spec:
  accessModes:
  - ReadWriteOnce
  storageClassName: $SC_NAME
  resources:
    requests:
      storage: 250Mi
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: $DEPLOYMENT
  namespace: $NAMESPACE
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mariadb
  template:
    metadata:
      labels:
        app: mariadb
    spec:
      containers:
      - name: mariadb
        image: mariadb:10.6
        env:
        - name: MYSQL_ROOT_PASSWORD
          value: "rootpassword"
        - name: MYSQL_DATABASE
          value: "appdb"
        ports:
        - containerPort: 3306
        volumeMounts:
        - name: mariadb-storage
          mountPath: /var/lib/mysql
      volumes:
      - name: mariadb-storage
        persistentVolumeClaim:
          claimName: $PVC_NAME
EOF

echo "  Waiting for MariaDB pod to be Running..."
kubectl wait --for=condition=available deployment/"$DEPLOYMENT" \
    -n "$NAMESPACE" --timeout=120s
echo "  MariaDB is Running."

# ── Step 6: Inject sample data ────────────────────────────────
echo "[6/7] Injecting sample data into MariaDB..."

POD=$(kubectl get pod -n "$NAMESPACE" -l app=mariadb \
    -o jsonpath='{.items[0].metadata.name}')

echo "  Waiting for MariaDB to accept connections..."
for i in $(seq 1 20); do
    if kubectl exec "$POD" -n "$NAMESPACE" -- \
        mysql -uroot -prootpassword -e "SELECT 1;" &>/dev/null; then
        echo "  MariaDB is accepting connections."
        break
    fi
    echo "  Waiting... ($i/20)"
    sleep 5
done

# Create table
kubectl exec "$POD" -n "$NAMESPACE" -- \
    mysql -uroot -prootpassword appdb \
    -e "CREATE TABLE IF NOT EXISTS employees (
        id INT AUTO_INCREMENT PRIMARY KEY,
        name VARCHAR(100),
        role VARCHAR(100),
        joined_date DATE
    );"

echo "  Table created."

# Truncate first to avoid duplicates on re-run
kubectl exec "$POD" -n "$NAMESPACE" -- \
    mysql -uroot -prootpassword appdb \
    -e "TRUNCATE TABLE employees;"
    
# Insert sample data
kubectl exec "$POD" -n "$NAMESPACE" -- \
    mysql -uroot -prootpassword appdb \
    -e "INSERT INTO employees (name, role, joined_date) VALUES
        ('Alice Johnson', 'DevOps Engineer', '2022-01-15'),
        ('Bob Smith', 'Backend Developer', '2021-06-20'),
        ('Carol White', 'SRE Lead', '2020-03-10');"

echo "  Sample data injected."

# Verify data
echo "  Verifying data..."
kubectl exec "$POD" -n "$NAMESPACE" -- \
    mysql -uroot -prootpassword appdb \
    -e "SELECT * FROM employees;"

# ── Step 7: Simulate accidental deletion ─────────────────────
echo "[7/7] Simulating accidental deletion..."

kubectl delete deployment "$DEPLOYMENT" -n "$NAMESPACE"
echo "  Deployment deleted."

kubectl delete pvc "$PVC_NAME" -n "$NAMESPACE"
echo "  PVC deleted."

echo "  Patching PV to remove claimRef (making it Available)..."
kubectl patch pv "$PV_NAME" \
    --type=json \
    -p='[{"op": "remove", "path": "/spec/claimRef"}]'

echo "  Waiting for PV to become Available..."
for i in $(seq 1 10); do
    STATUS=$(kubectl get pv "$PV_NAME" \
        -o jsonpath='{.status.phase}')
    if [ "$STATUS" = "Available" ]; then
        echo "  PV is Available."
        break
    fi
    echo "  Waiting... ($i/10)"
    sleep 3
done

# Drop deployment YAML file without PVC
cat > "$DEPLOYMENT_FILE" <<EOF
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mariadb
  namespace: mariadb
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mariadb
  template:
    metadata:
      labels:
        app: mariadb
    spec:
      containers:
      - name: mariadb
        image: mariadb:10.6
        env:
        - name: MYSQL_ROOT_PASSWORD
          value: "rootpassword"
        - name: MYSQL_DATABASE
          value: "appdb"
        ports:
        - containerPort: 3306
EOF

echo "  Deployment YAML created (no PVC configured)."
echo ""
kubectl get pv "$PV_NAME"
echo ""
echo "✅ Practice scenario is ready!"
echo "   PV exists with data intact."
echo "   Deployment and PVC have been deleted."
echo "   Deployment YAML is at ~/mariadb-deployment.yaml"
echo "   Go restore the Deployment and recover the data! 💪"
echo ""
```

# ✅ Solution

<details>
<summary>Click to expand the solution</summary>

### Step 1: Check the existing PersistentVolume

```bash
kubectl get pv
```

Note the **storageClassName**, **capacity** and **accessModes** — PVC must match these to bind to this PV.

```
NAME         CAPACITY   ACCESS MODES   RECLAIM POLICY   STATUS      STORAGECLASS   CLAIM
mariadb-pv   250Mi      RWO            Retain           Available   local-path
```

---

### Step 2: Create the PVC manifest

```bash
vim mariadb-pvc.yaml
```

```yaml
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: mariadb
  namespace: mariadb
spec:
  accessModes:
  - ReadWriteOnce
  storageClassName: local-path
  resources:
    requests:
      storage: 250Mi
```

`Note`: 💡 Since only one PV exists with storageClassName: local-path, the PVC will automatically bind to it. If multiple PVs existed, use volumeName: mariadb-pv in the PVC spec to target a specific PV.

```bash
kubectl apply -f mariadb-pvc.yaml
```

---

### Step 3: Verify PVC is bound to the PV

```bash
kubectl get pvc mariadb -n mariadb
```

Expected output:

```
NAME      STATUS   VOLUME       CAPACITY   ACCESS MODES   STORAGECLASS   AGE
mariadb   Bound    mariadb-pv   250Mi      RWO            local-path     10s
```

Status must show `Bound` before proceeding.

---

### Step 4: Edit the Deployment YAML to add PVC

```bash
vim ~/mariadb-deployment.yaml
```

Add `volumeMounts` to the container and `volumes` to the pod spec:

```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mariadb
  namespace: mariadb
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mariadb
  template:
    metadata:
      labels:
        app: mariadb
    spec:
      containers:
      - name: mariadb
        image: mariadb:10.6
        env:
        - name: MYSQL_ROOT_PASSWORD
          value: "rootpassword"
        - name: MYSQL_DATABASE
          value: "appdb"
        ports:
        - containerPort: 3306
        volumeMounts:                   # Add this block
        - name: mariadb-storage
          mountPath: /var/lib/mysql
      volumes:                          # Add this block
      - name: mariadb-storage
        persistentVolumeClaim:
          claimName: mariadb
```

Save and exit.

---

### Step 5: Apply the updated Deployment

```bash
kubectl apply -f ~/mariadb-deployment.yaml
```

---

### Step 6: Verify the Deployment is running and stable

```bash
kubectl rollout status deployment/mariadb -n mariadb
```

```bash
kubectl get pods -n mariadb
```

Expected output:

```
NAME                       READY   STATUS    RESTARTS   AGE
mariadb-xxxxxxxxx-xxxxx    1/1     Running   0          30s
```

---

### Step 7: Verify data is intact

```bash
POD=$(kubectl get pod -n mariadb -l app=mariadb \
    -o jsonpath='{.items[0].metadata.name}')

kubectl exec "$POD" -n mariadb -- \
    mysql -uroot -prootpassword appdb \
    -e "SELECT * FROM employees;"
```

Expected output — all 3 records should be present:

```
+----+---------------+-------------------+-------------+
| id | name          | role              | joined_date |
+----+---------------+-------------------+-------------+
|  1 | Alice Johnson | DevOps Engineer   | 2022-01-15  |
|  2 | Bob Smith     | Backend Developer | 2021-06-20  |
|  3 | Carol White   | SRE Lead          | 2020-03-10  |
+----+---------------+-------------------+-------------+
```

If all 3 records are visible — data persistence is confirmed! ✅

### 🧹 Script : reset_q02_pvc.sh

```bash
#!/bin/bash
# ============================================================
# CKA Practice | q02: PVC and Deployment Data Persistence
# reset_q02_pvc.sh
# Removes namespace, PV, StorageClass (if created by setup),
# deployment file, candidate files and hostPath directory
# Run on: controlplane node
# ============================================================

set -euo pipefail

NAMESPACE="mariadb"
PV_NAME="mariadb-pv"
SC_NAME="local-path"
DEPLOYMENT_FILE="$HOME/mariadb-deployment.yaml"
HOST_PATH="/mnt/mariadb-data"

echo ""
echo "==========================================="
echo " CKA q02 — Restoring cluster to baseline..."
echo "==========================================="
echo ""

# ── Delete namespace ──────────────────────────────────────────
echo "[1/6] Deleting namespace '$NAMESPACE'..."
kubectl delete namespace "$NAMESPACE" --ignore-not-found
echo "  Namespace and all resources removed."

# ── Delete PV ─────────────────────────────────────────────────
echo "[2/6] Deleting PersistentVolume '$PV_NAME'..."
kubectl delete pv "$PV_NAME" --ignore-not-found
echo "  PersistentVolume deleted."

# ── Delete StorageClass only if it uses no-provisioner ────────
# (avoid deleting pre-existing cluster StorageClass)
echo "[3/6] Checking StorageClass '$SC_NAME'..."
SC_PROVISIONER=$(kubectl get storageclass "$SC_NAME" \
    -o jsonpath='{.provisioner}' 2>/dev/null || true)
if [ "$SC_PROVISIONER" = "kubernetes.io/no-provisioner" ]; then
    kubectl delete storageclass "$SC_NAME" --ignore-not-found
    echo "  StorageClass deleted."
else
    echo "  StorageClass is pre-existing cluster SC — skipping deletion."
fi

# ── Remove deployment file and candidate YAML files ───────────
echo "[4/6] Removing deployment and candidate YAML files..."
rm -f "$DEPLOYMENT_FILE"
rm -f "$HOME/mariadb-pvc.yaml"
echo "  Files removed."

# ── Remove host path directory ────────────────────────────────
echo "[5/6] Removing host path directory '$HOST_PATH'..."
sudo rm -rf "$HOST_PATH"
echo "  Host path directory removed."

echo ""
echo "✅ Cluster is back to baseline!"
echo "   Safe to move to next problem. 🚀"
echo ""
```
