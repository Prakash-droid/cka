# 🧩 Problem Statement (Configure cri-dockerd for Kubernetes)

<details>
<summary>Click to expand the problem statement</summary>

A Linux system has Docker installed but needs to be configured for Kubernetes using kubeadm.

Your task is to:

1. Connect to the **correct host**
2. Install and configure **cri-dockerd**
3. Configure required **system kernel parameters** that persist across reboots

```
[candidate@base]$ ssh <hostname>
```

### 📋 Configure the following:

1. Set up **cri-dockerd**:
   * Install the Debian package: **`~/cri-dockerd_0.3.6.3-0.ubuntu-jammy_amd64.deb`**
   * Use `dpkg` to install the package
   * Enable and start the **`cri-docker`** service

2. Configure the following system parameters — ensure they **persist across reboots** and apply to the **running system**:
   * `net.bridge.bridge-nf-call-iptables = 1`
   * `net.ipv6.conf.all.forwarding = 1`
   * `net.ipv4.ip_forward = 1`
   * `net.netfilter.nf_conntrack_max = 131072`

### 📥 Script : setup_q04_cridockerd.sh

```bash
#!/bin/bash
# ============================================================
# CKA Practice | Q04: Configure cri-dockerd for Kubernetes
# setup_q04_cridockerd.sh
# Downloads cri-dockerd deb package, resets sysctl params
# and removes cri-docker service if previously installed
# Re-run safe: only resets candidate work on subsequent runs
# Run on: controlplane node
# ============================================================

set -euo pipefail

CRI_DOCKERD_VERSION="0.3.6.3-0"
CRI_DOCKERD_DEB="cri-dockerd_${CRI_DOCKERD_VERSION}.ubuntu-jammy_amd64.deb"
CRI_DOCKERD_URL="https://github.com/Mirantis/cri-dockerd/releases/download/v0.3.6/${CRI_DOCKERD_DEB}"
SYSCTL_FILE="/etc/sysctl.d/99-kubernetes.conf"

echo ""
echo "==========================================="
echo " CKA Q04 — Setting up cri-dockerd scenario..."
echo "==========================================="
echo ""

# ── Step 1: Reset candidate work ─────────────────────────────
echo "[1/5] Resetting candidate work from previous attempt..."

# Stop and disable cri-docker service if running
if systemctl is-active cri-docker &>/dev/null; then
    echo "  Stopping cri-docker service..."
    sudo systemctl stop cri-docker 2>/dev/null || true
    sudo systemctl stop cri-docker.socket 2>/dev/null || true
fi

if systemctl is-enabled cri-docker &>/dev/null; then
    echo "  Disabling cri-docker service..."
    sudo systemctl disable cri-docker 2>/dev/null || true
    sudo systemctl disable cri-docker.socket 2>/dev/null || true
fi

# Uninstall cri-dockerd if installed
if dpkg -l cri-dockerd &>/dev/null; then
    echo "  Uninstalling cri-dockerd..."
    sudo dpkg -r cri-dockerd 2>/dev/null || true
    echo "  cri-dockerd uninstalled."
fi

# Remove sysctl config file if exists
if [ -f "$SYSCTL_FILE" ]; then
    sudo rm -f "$SYSCTL_FILE"
    echo "  sysctl config file removed."
fi

# Reset sysctl params to defaults
echo "  Resetting sysctl parameters to defaults..."
sudo sysctl -w net.bridge.bridge-nf-call-iptables=0 2>/dev/null || true
sudo sysctl -w net.ipv6.conf.all.forwarding=0 2>/dev/null || true
sudo sysctl -w net.ipv4.ip_forward=0 2>/dev/null || true
sudo sysctl -w net.netfilter.nf_conntrack_max=65536 2>/dev/null || true

echo "  Candidate work cleared."

# ── Step 2: Check Docker installation (official repo method) ─
echo "[2/5] Checking Docker installation..."

# Check if docker-ce is installed (official repo method)
# docker.io is NOT acceptable — must be docker-ce from Docker's official repo
DOCKER_INSTALLED=false
if dpkg -l docker-ce &>/dev/null 2>&1; then
    echo "  Docker CE already installed (official repo): $(docker --version)"
    DOCKER_INSTALLED=true
fi

if [ "$DOCKER_INSTALLED" = false ]; then
    echo "  Docker CE not found — installing from official Docker repo..."

    # Install dependencies
    sudo apt-get update -qq
    sudo apt-get install -y ca-certificates curl

    # Add Docker's official GPG key
    sudo install -m 0755 -d /etc/apt/keyrings
    sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg \
        -o /etc/apt/keyrings/docker.asc
    sudo chmod a+r /etc/apt/keyrings/docker.asc

    # Add Docker official repository
    sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF

    sudo apt-get update -qq

    # Install Docker CE (official packages)
    # DEBIAN_FRONTEND=noninteractive — suppresses all interactive prompts
    # -o Dpkg::Options::="--force-confnew" — always use package maintainer's config file
    sudo DEBIAN_FRONTEND=noninteractive apt-get install -y \
        -o Dpkg::Options::="--force-confdef" \
        -o Dpkg::Options::="--force-confnew" \
        docker-ce \
        docker-ce-cli \
        containerd.io \
        docker-buildx-plugin \
        docker-compose-plugin

    sudo systemctl enable docker
    sudo systemctl start docker

    echo "  Docker CE installed successfully: $(docker --version)"
fi

# Verify Docker is active
if ! systemctl is-active docker &>/dev/null; then
    echo "  Docker service not running — starting..."
    sudo systemctl start docker
fi

# ── Step 3: Download cri-dockerd package ─────────────────────
echo "[3/5] Downloading cri-dockerd package..."
if [ -f "$HOME/$CRI_DOCKERD_DEB" ]; then
    echo "  Package already exists — skipping download."
else
    echo "  Downloading $CRI_DOCKERD_DEB..."
    curl -L "$CRI_DOCKERD_URL" -o "$HOME/$CRI_DOCKERD_DEB"
    echo "  Package downloaded."
fi

# ── Step 4: Verify setup ──────────────────────────────────────
echo "[4/5] Verifying setup..."
echo ""
ls -lh "$HOME/$CRI_DOCKERD_DEB"
echo ""
echo "  Docker status:"
systemctl is-active docker || true
echo "  Docker version:"
docker --version
echo ""
echo "  Current sysctl values (should be 0 — reset done):"
sysctl net.bridge.bridge-nf-call-iptables 2>/dev/null || true
sysctl net.ipv4.ip_forward 2>/dev/null || true
echo ""

# ── Step 5: Final summary ─────────────────────────────────────
echo "[5/5] Setup complete!"
echo ""
echo "✅ Practice scenario is ready!"
echo ""
echo "   cri-dockerd package is at: ~/$CRI_DOCKERD_DEB"
echo "   Docker CE is installed and running (official repo)."
echo "   sysctl params are reset to defaults."
echo "   💪 Go install cri-dockerd and configure sysctl!"
echo ""
```

# ✅ Solution

<details>
<summary>Click to expand the solution</summary>

### Step 1: Install cri-dockerd using dpkg

```bash
sudo dpkg -i ~/cri-dockerd_0.3.6.3-0.ubuntu-jammy_amd64.deb
```

Expected output:

```
Selecting previously unselected package cri-dockerd.
(Reading database ...)
Preparing to unpack .../cri-dockerd_0.3.6.3-0.ubuntu-jammy_amd64.deb ...
Unpacking cri-dockerd ...
Setting up cri-dockerd ...
```

---

### Step 2: Enable and start the cri-docker service

```bash
sudo systemctl enable cri-docker
sudo systemctl start cri-docker
```

Verify the service is running:

```bash
sudo systemctl status cri-docker
```

Expected output:

```
● cri-docker.service - CRI Interface for Docker Application Container Engine
   Loaded: loaded (/lib/systemd/system/cri-docker.service; enabled)
   Active: active (running)
```

---

### Step 3: Load required kernel modules

Before applying sysctl params, ensure `br_netfilter` and `nf_conntrack` modules are loaded:

```bash
sudo modprobe br_netfilter
sudo modprobe nf_conntrack
```

Make them persist across reboots:

```bash
cat <<EOF | sudo tee /etc/modules-load.d/kubernetes.conf
br_netfilter
nf_conntrack
EOF
```

---

### Step 4: Configure sysctl parameters

Create a persistent sysctl config file:

```bash
cat <<EOF | sudo tee /etc/sysctl.d/99-kubernetes.conf
net.bridge.bridge-nf-call-iptables = 1
net.ipv6.conf.all.forwarding = 1
net.ipv4.ip_forward = 1
net.netfilter.nf_conntrack_max = 131072
EOF
```

---

### Step 5: Apply sysctl parameters to the running system

```bash
sudo sysctl --system
```

Expected output — showing parameters being applied:

```
* Applying /etc/sysctl.d/99-kubernetes.conf ...
net.bridge.bridge-nf-call-iptables = 1
net.ipv6.conf.all.forwarding = 1
net.ipv4.ip_forward = 1
net.netfilter.nf_conntrack_max = 131072
```

---

### Step 6: Verify all parameters are applied

```bash
sysctl net.bridge.bridge-nf-call-iptables \
       net.ipv6.conf.all.forwarding \
       net.ipv4.ip_forward \
       net.netfilter.nf_conntrack_max
```

Expected output:

```
net.bridge.bridge-nf-call-iptables = 1
net.ipv6.conf.all.forwarding = 1
net.ipv4.ip_forward = 1
net.netfilter.nf_conntrack_max = 131072
```

---

### Step 7: Verify cri-docker service is enabled and running

```bash
sudo systemctl is-enabled cri-docker
sudo systemctl is-active cri-docker
```

Expected output:

```
enabled
active
```

All tasks complete. ✅

### 🧹 Script : reset_q04_cridockerd.sh

```bash
#!/bin/bash
# ============================================================
# CKA Practice | Q04: Configure cri-dockerd for Kubernetes
# reset_q04_cridockerd.sh
# Removes cri-dockerd, sysctl config and resets params
# Run on: controlplane node
# ============================================================

set -euo pipefail

SYSCTL_FILE="/etc/sysctl.d/99-kubernetes.conf"
MODULES_FILE="/etc/modules-load.d/kubernetes.conf"

echo ""
echo "==========================================="
echo " CKA Q04 — Restoring cluster to baseline..."
echo "==========================================="
echo ""

# ── Stop and disable cri-docker service ──────────────────────
echo "[1/4] Stopping and disabling cri-docker service..."
sudo systemctl stop cri-docker 2>/dev/null || true
sudo systemctl stop cri-docker.socket 2>/dev/null || true
sudo systemctl disable cri-docker 2>/dev/null || true
sudo systemctl disable cri-docker.socket 2>/dev/null || true
echo "  cri-docker service stopped and disabled."

# ── Uninstall cri-dockerd ─────────────────────────────────────
echo "[2/4] Uninstalling cri-dockerd..."
if dpkg -l cri-dockerd &>/dev/null; then
    sudo dpkg -r cri-dockerd 2>/dev/null || true
    echo "  cri-dockerd uninstalled."
else
    echo "  cri-dockerd not installed — skipping."
fi

# ── Remove sysctl and modules config files ────────────────────
echo "[3/4] Removing config files..."
sudo rm -f "$SYSCTL_FILE"
sudo rm -f "$MODULES_FILE"
echo "  Config files removed."

# ── Reset sysctl parameters to defaults ──────────────────────
echo "[4/4] Resetting sysctl parameters to defaults..."
sudo sysctl -w net.bridge.bridge-nf-call-iptables=0 2>/dev/null || true
sudo sysctl -w net.ipv6.conf.all.forwarding=0 2>/dev/null || true
sudo sysctl -w net.ipv4.ip_forward=0 2>/dev/null || true
sudo sysctl -w net.netfilter.nf_conntrack_max=65536 2>/dev/null || true
echo "  sysctl parameters reset."

echo ""
echo "✅ Cluster is back to baseline!"
echo "   Safe to move to next problem. 🚀"
echo ""
```
